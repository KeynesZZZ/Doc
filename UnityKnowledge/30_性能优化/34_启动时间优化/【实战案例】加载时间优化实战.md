## 3. 解决方案

### 3.1 迁移到 Addressables

**问题**：Resources.Load 同步加载阻塞主线程

**方案**：使用 Addressables 异步加载

> **前置条件**：使用任何 Addressables API 前，必须确保初始化完成。`Addressables.InitializeAsync()` 在首次调用 `LoadAssetAsync` 时自动触发（惰性初始化），但建议在启动流程中显式调用以避免不可预期的延迟。
>
> ```csharp
> // 启动时显式初始化 Addressables
> var initHandle = Addressables.InitializeAsync();
> await initHandle.Task;
> // 初始化完成后才加载资源
> ```
>
> 初始化内容包括：加载 Catalog（资源地址映射表）、初始化 Resource Manager、检查缓存目录。Catalog 解析可能耗时 50-200ms（取决于资源数量），在 Splash Screen 或加载界面阶段完成即可。

#### 步骤1：资源迁移

```csharp
/// <summary>
/// 资源加载服务 - Addressables封装
/// </summary>
public class AssetLoader : MonoBehaviour
{
    private Dictionary<string, AsyncOperationHandle> loadedHandles = new Dictionary<string, AsyncOperationHandle>();

    /// <summary>
    /// 异步加载单个资源
    /// </summary>
    public async Task<T> LoadAssetAsync<T>(string address) where T : Object
    {
        // 检查缓存
        if (loadedHandles.TryGetValue(address, out var existingHandle))
        {
            if (existingHandle.IsDone)
            {
                return existingHandle.Result as T;
            }
            await existingHandle.Task;
            return existingHandle.Result as T;
        }

        // 异步加载
        var handle = Addressables.LoadAssetAsync<T>(address);
        loadedHandles[address] = handle;

        await handle.Task;

        if (handle.Status == AsyncOperationStatus.Succeeded)
        {
            return handle.Result;
        }

        Debug.LogError($"Failed to load asset: {address}");
        return null;
    }

    /// <summary>
    /// 批量异步加载
    /// </summary>
    public async Task<List<T>> LoadAssetsAsync<T>(IList<string> addresses) where T : Object
    {
        var results = new List<T>();
        var tasks = new List<Task<T>>();

        foreach (var address in addresses)
        {
            tasks.Add(LoadAssetAsync<T>(address));
        }

        var loadedAssets = await Task.WhenAll(tasks);
        results.AddRange(loadedAssets.Where(a => a != null));

        return results;
    }

    /// <summary>
    /// 释放资源
    /// </summary>
    public void ReleaseAsset(string address)
    {
        if (loadedHandles.TryGetValue(address, out var handle))
        {
            Addressables.Release(handle);
            loadedHandles.Remove(address);
        }
    }

    /// <summary>
    /// 释放所有资源
    /// </summary>
    public void ReleaseAll()
    {
        foreach (var handle in loadedHandles.Values)
        {
            Addressables.Release(handle);
        }
        loadedHandles.Clear();
    }
}
```

#### 步骤2：重构UI资源加载

```csharp
/// <summary>
/// UI管理器 - 异步加载版本
/// </summary>
public class UIManager : MonoBehaviour
{
    [SerializeField] private AssetLoader assetLoader;

    private Dictionary<string, GameObject> cachedPanels = new Dictionary<string, GameObject>();

    /// <summary>
    /// 异步打开面板
    /// </summary>
    public async Task<T> OpenPanelAsync<T>(string panelAddress) where T : Component
    {
        // 检查缓存
        if (cachedPanels.TryGetValue(panelAddress, out var cachedPanel))
        {
            cachedPanel.SetActive(true);
            return cachedPanel.GetComponent<T>();
        }

        // 异步加载
        var prefab = await assetLoader.LoadAssetAsync<GameObject>(panelAddress);
        if (prefab == null) return null;

        // 实例化
        var panel = Instantiate(prefab, transform);
        cachedPanels[panelAddress] = panel;

        return panel.GetComponent<T>();
    }

    /// <summary>
    /// 关闭面板（回收到缓存）
    /// </summary>
    public void ClosePanel(string panelAddress)
    {
        if (cachedPanels.TryGetValue(panelAddress, out var panel))
        {
            panel.SetActive(false);
        }
    }
}
```

### 3.2 异步场景加载

**问题**：SceneManager.LoadScene 同步阻塞

**方案**：使用 LoadSceneAsync + 进度显示

```csharp
/// <summary>
/// 场景管理器 - 异步加载版本
/// </summary>
public class SceneLoader : MonoBehaviour
{
    [Header("Loading UI")]
    [SerializeField] private GameObject loadingScreen;
    [SerializeField] private Slider progressBar;
    [SerializeField] private Text progressText;

    private AsyncOperation currentOperation;

    /// <summary>
    /// 异步加载场景（带进度显示）
    /// </summary>
    public async Task LoadSceneAsync(string sceneName)
    {
        // 显示加载界面
        loadingScreen.SetActive(true);
        UpdateProgress(0f);

        // 开始异步加载
        currentOperation = SceneManager.LoadSceneAsync(sceneName);
        currentOperation.allowSceneActivation = false;

        // 更新进度
        // 注意: allowSceneActivation=false 时，progress 最大到 0.9 且 isDone 永远为 false
        // 必须手动设置 allowSceneActivation=true 才能完成场景激活
        while (currentOperation.progress < 0.9f)
        {
            float progress = Mathf.Clamp01(currentOperation.progress / 0.9f);
            UpdateProgress(progress);
            await Task.Yield();
        }

        // 加载完成（progress 到达 0.9），显示 100% 并激活场景
        UpdateProgress(1f);
        await Task.Delay(100); // 短暂延迟让用户看到 100%
        currentOperation.allowSceneActivation = true;

        // 等待场景激活完成
        while (!currentOperation.isDone)
        {
            await Task.Yield();
        }

        // 隐藏加载界面
        loadingScreen.SetActive(false);
    }

    /// <summary>
    /// 带预加载的场景切换
    /// </summary>
    public async Task LoadSceneWithPreload(string sceneName, List<string> preloadAssets)
    {
        loadingScreen.SetActive(true);
        UpdateProgress(0f);

        // 阶段1: 加载场景 (0% - 50%)
        var sceneTask = LoadSceneInternal(sceneName);
        var progressTask = MonitorSceneProgress();

        await sceneTask;

        // 阶段2: 预加载资源 (50% - 100%)
        UpdateProgress(0.5f);
        await PreloadAssets(preloadAssets, (p) => UpdateProgress(0.5f + p * 0.5f));

        UpdateProgress(1f);
        await Task.Delay(100);

        loadingScreen.SetActive(false);
    }

    private void UpdateProgress(float progress)
    {
        if (progressBar != null)
        {
            progressBar.value = progress;
        }
        if (progressText != null)
        {
            progressText.text = $"加载中... {(int)(progress * 100)}%";
        }
    }
}
```

### 3.3 分帧初始化

**问题**：所有Manager在Awake/Start同步初始化导致卡顿

**方案**：使用协程分帧初始化

```csharp
/// <summary>
/// 游戏启动器 - 分帧初始化
/// </summary>
public class GameLauncher : MonoBehaviour
{
    [Header("Initialization Order")]
    [SerializeField] private List<MonoBehaviour> initOrder;

    [Header("Loading UI")]
    [SerializeField] private Slider initProgress;
    [SerializeField] private Text initStatus;

    private float initStartTime;

    private async void Start()
    {
        initStartTime = Time.realtimeSinceStartup;

        // 阶段1: 核心系统初始化
        await InitializePhase("核心系统", new List<Func<Task>>
        {
            () => InitializeManager<AudioManager>(),
            () => InitializeManager<EventManager>(),
            () => InitializeManager<SaveManager>(),
        });

        // 阶段2: UI初始化
        await InitializePhase("UI系统", new List<Func<Task>>
        {
            () => InitializeManager<UIManager>(),
            () => PreloadUIAssets(),
        });

        // 阶段3: 游戏系统初始化
        await InitializePhase("游戏系统", new List<Func<Task>>
        {
            () => InitializeManager<GameManager>(),
            () => InitializeManager<CharacterManager>(),
            () => LoadLevelConfig(),
        });

        // 阶段4: 可延后初始化
        StartCoroutine(DelayedInit());
    }

    private async Task InitializePhase(string phaseName, List<Func<Task>> tasks)
    {
        UpdateStatus($"正在初始化{phaseName}...");

        for (int i = 0; i < tasks.Count; i++)
        {
            await tasks[i]();
            UpdateProgress((float)(i + 1) / tasks.Count);

            // 每个任务后等待一帧，避免卡顿
            await Task.Yield();
        }
    }

    private IEnumerator DelayedInit()
    {
        // 等待3秒后初始化非关键系统
        yield return new WaitForSeconds(3f);

        // 延迟初始化分析SDK（在主线程上，避免卡顿即可）
        // 注意: 大多数 SDK 必须在主线程初始化，不能用 Task.Run
        AnalyticsService.Initialize();

        yield return new WaitForSeconds(0.5f);

        // 延迟初始化广告SDK
        AdService.Initialize();
    }

    private void UpdateStatus(string status)
    {
        if (initStatus != null)
        {
            initStatus.text = status;
        }
    }

    private void UpdateProgress(float progress)
    {
        if (initProgress != null)
        {
            initProgress.value = progress;
        }
    }

    private async Task InitializeManager<T>() where T : MonoBehaviour
    {
        var manager = FindObjectOfType<T>();
        if (manager is IAsyncInitializable asyncInit)
        {
            await asyncInit.InitializeAsync();
        }
    }
}

/// <summary>
/// 异步初始化接口
/// </summary>
public interface IAsyncInitializable
{
    Task InitializeAsync();
}
```

### 3.4 配置预加载

**问题**：JSON配置在主线程解析阻塞

**方案**：后台线程预加载 + 缓存

```csharp
/// <summary>
/// 配置管理器 - 异步加载版本
/// </summary>
public class ConfigManager : MonoBehaviour, IAsyncInitializable
{
    private Dictionary<string, object> configCache = new Dictionary<string, object>();

    public async Task InitializeAsync()
    {
        // 在后台线程加载所有配置
        var configFiles = new Dictionary<string, Type>
        {
            { "Config/levels", typeof(LevelConfigData) },
            { "Config/characters", typeof(CharacterConfigData) },
            { "Config/items", typeof(ItemConfigData) },
            { "Config/iap", typeof(IAPConfigData) },
        };

        // 并行加载
        var tasks = configFiles.Select(kvp => LoadConfigAsync(kvp.Key, kvp.Value));
        await Task.WhenAll(tasks);
    }

    private async Task LoadConfigAsync(string address, Type configType)
    {
        var jsonAsset = await Addressables.LoadAssetAsync<TextAsset>(address).Task;

        if (jsonAsset != null)
        {
            // 在后台线程解析JSON
            var config = await Task.Run(() =>
            {
                return JsonUtility.FromJson(jsonAsset.text, configType);
            });

            configCache[address] = config;
        }
    }

    /// <summary>
    /// 获取配置（同步，必须已预加载）
    /// </summary>
    public T GetConfig<T>(string address) where T : class
    {
        if (configCache.TryGetValue(address, out var config))
        {
            return config as T;
        }
        Debug.LogError($"Config not loaded: {address}");
        return null;
    }
}
```

### 3.5 智能预加载策略

**问题**：全部资源一次性加载太慢

**方案**：按优先级分层预加载

```csharp
/// <summary>
/// 智能预加载管理器
/// </summary>
public class SmartPreloader : MonoBehaviour
{
    [System.Serializable]
    public class PreloadGroup
    {
        public string groupName;
        public List<string> assetAddresses;
        public PreloadPriority priority;
        public bool showInLoading;
    }

    public enum PreloadPriority
    {
        Critical,   // 启动必须，立即加载
        High,       // 首屏需要，启动后加载
        Normal,     // 游戏中需要，后台加载
        Low         // 可选内容，空闲加载
    }

    [Header("Preload Groups")]
    [SerializeField] private List<PreloadGroup> preloadGroups;

    private Dictionary<string, float> loadProgress = new Dictionary<string, float>();

    /// <summary>
    /// 开始预加载
    /// </summary>
    public async Task StartPreload(Action<float> onProgress)
    {
        // 按优先级分组
        var criticalGroup = preloadGroups.Where(g => g.priority == PreloadPriority.Critical);
        var highGroup = preloadGroups.Where(g => g.priority == PreloadPriority.High);
        var normalGroup = preloadGroups.Where(g => g.priority == PreloadPriority.Normal);
        var lowGroup = preloadGroups.Where(g => g.priority == PreloadPriority.Low);

        int totalGroups = preloadGroups.Count;
        int completedGroups = 0;

        // 1. 加载关键资源（阻塞）
        foreach (var group in criticalGroup)
        {
            await LoadGroup(group);
            completedGroups++;
            onProgress?.Invoke((float)completedGroups / totalGroups);
        }

        // 2. 启动游戏（关键资源加载完成即可）
        // ... 允许玩家进入游戏 ...

        // 3. 后台加载高优先级资源（用协程，避免线程安全问题）
        // 注意: Addressables API 必须在主线程调用，不能用 Task.Run
        StartCoroutine(BackgroundPreload(highGroup, normalGroup, lowGroup,
            totalGroups, completedGroups, onProgress));
    }

    private async Task LoadGroup(PreloadGroup group)
    {
        var tasks = group.assetAddresses.Select(addr =>
            Addressables.LoadAssetAsync<Object>(addr).Task);

        await Task.WhenAll(tasks);
    }

    /// <summary>
    /// 后台预加载协程（Addressables 必须在主线程调用）
    /// </summary>
    private IEnumerator BackgroundPreload(
        IEnumerable<PreloadGroup> high,
        IEnumerable<PreloadGroup> normal,
        IEnumerable<PreloadGroup> low,
        int totalGroups, int completedGroups, Action<float> onProgress)
    {
        foreach (var group in high)
        {
            yield return LoadGroupCoroutine(group);
            completedGroups++;
            onProgress?.Invoke((float)completedGroups / totalGroups);
        }

        foreach (var group in normal)
        {
            yield return LoadGroupCoroutine(group);
            completedGroups++;
        }

        foreach (var group in low)
        {
            yield return LoadGroupCoroutine(group);
        }
    }

    private IEnumerator LoadGroupCoroutine(PreloadGroup group)
    {
        foreach (var addr in group.assetAddresses)
        {
            var handle = Addressables.LoadAssetAsync<Object>(addr);
            yield return handle;

            if (handle.Status != AsyncOperationStatus.Succeeded)
                Debug.LogWarning($"[Preloader] 加载失败: {addr}");
        }
    }
}
```

#### Shader Variant Collection 具体操作步骤

**第1步：确认需要预编译哪些 Shader Variant**

```
1. 打开 Profiler（Window > Analysis > Profiler）
2. 切换到 CPU Usage 模块 → Hierarchy 视图
3. 过滤 "Shader" 关键字
4. 运行游戏，观察 Shader.CreateGPUProgram / Shader.Parse 调用
5. 记录每个 Shader 编译了哪些 Variant（Pass + Keyword 组合）
```

**第2步：创建 Shader Variant Collection**

```
1. Project 窗口右键 → Create → Shader Variant Collection
2. 命名如 "SVC_URP_Main"
3. 在 Inspector 中点 "+" 添加 Shader
4. 选择项目中使用的 Shader（如 URP/Lit、UI/Default 等）
5. 展开每个 Shader，勾选实际使用的 Variant:
   ├─ Pass: ForwardLit / ShadowCaster / DepthOnly 等
   ├─ Keywords: _NORMALMAP / _EMISSION / _ALPHATEST_ON 等
   └─ 只勾实际用到的组合，不要全选（否则包体膨胀）
```

**第3步：配置到 Player Settings 预加载**

```
Project Settings > Player > Other Settings >
  ├─ Shader Variant Collection: 拖入 SVC_URP_Main.asset
  └─ 这些 Variant 会在引擎初始化时预编译（Splash Screen 阶段完成）
```

或通过代码在加载界面预热：

```csharp
/// <summary>
/// Shader 预热器：在加载界面期间编译所有 Shader Variant
/// </summary>
public class ShaderWarmupper : MonoBehaviour
{
    [SerializeField] private ShaderVariantCollection variantCollection;

    /// <summary>
    /// 预热所有 Shader Variant（在加载界面调用）
    /// </summary>
    public IEnumerator Warmup()
    {
        if (variantCollection == null) yield break;

        // 逐个编译，每编译几个 yield 一次避免卡顿
        int count = 0;
        foreach (var variant in variantCollection.shaders)
        {
            foreach (var info in variant.variants)
            {
                // 记录编译前的 Shader 数量
                int before = Shader.globalRenderPlatformCount;
                variantCollection.Add(new ShaderVariantCollection.ShaderVariant(
                    info.shader, info.passType, info.keywords));
                count++;

                // 每5个 yield 一次
                if (count % 5 == 0)
                    yield return null;
            }
        }

        // 统一编译
        variantCollection.WarmUp();
        Debug.Log($"[Shader] 预热完成: {count} 个 Variant");
    }
}
```

**第4步：验证效果**

```

  Shader.CreateGPUProgram  × 127 次  总计 1023ms

优化后 Profiler:
  Shader.CreateGPUProgram  × 127 次  总计 98ms (在 Splash Screen 阶段完成)

```

## 6. 踩坑经验

> 以下是实际项目中踩过的三个典型坑及解决方案。

### 坑1：异步加载的进度不准确

```csharp
// ❌ 问题：Unity的AsyncOperation.progress在allowSceneActivation=false时最大值是0.9
while (loadOp.progress < 1f)
{
    // 永远等不到1f
    yield return null;
}

// ✅ 解决：检查isDone，或用 progress / 0.9f 映射
while (!loadOp.isDone)
{
    float normalizedProgress = Mathf.Clamp01(loadOp.progress / 0.9f);
    UpdateProgress(normalizedProgress);
    yield return null;
}
```

### 坑2：过早加载非关键资源

```csharp
// ❌ 问题：在启动时加载商城资源
void Start()
{
    LoadShopAssets();  // 其实不需要立即加载
}

// ✅ 解决：延迟到进入商城时加载
void OnEnterShop()
{
    StartCoroutine(LoadShopAssets());
}
```

### 坑3：忽略资源释放

```csharp
// ❌ 问题：资源只加载不释放，内存持续增长
void LoadScene()
{
    Resources.Load<GameObject>("Prefab");
    // 从不调用Unload
}

// ✅ 解决：及时释放资源
void UnloadScene()
{
    Resources.UnloadUnusedAssets();
    // 或使用 Addressables.Release(handle)
}
```
